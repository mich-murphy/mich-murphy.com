{{- /* Hugo 0.166's embedded link hook: a destination resolves to a page, then a page resource, then a global resource. An internal destination (a path with no scheme or host, like "/posts/x.md") that resolves to none of them fails the build, so a broken link can't ship */ -}} {{- $u := urls.Parse .Destination -}} {{- $href := $u.String -}} {{- if strings.HasPrefix $u.String "#" -}} {{- $href = printf "%s#%s" .PageInner.RelPermalink $u.Fragment -}} {{- else if and $href (not $u.IsAbs) -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or ($.PageInner.GetPage $path) ($.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $href = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $href = printf "%s?%s" $href . -}} {{- end -}} {{- with $u.Fragment -}} {{- $href = printf "%s#%s" $href . -}} {{- end -}} {{- else -}} {{- if and $path (not $u.Host) -}} {{- errorf "link to %q in %s doesn't resolve to a page or resource; link to the content file (e.g. /posts/.md) or use a full https:// URL" $.Destination (partial "hook-source.html" $) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* print writes a link's address after it (main.css), unless its text already is the address, as an autolink's is, or the link is within the page, whose address the header prints. Class no-print-url marks those, comparing with or without the scheme and a trailing slash */ -}} {{- $bare := `^https?://|/$` -}} {{- $addr := or (strings.HasPrefix .Destination "#") (eq (replaceRE $bare "" .PlainText) (replaceRE $bare "" .Destination)) -}} {{ .Text }} {{- /**/ -}}
