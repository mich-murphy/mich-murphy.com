{{- /* A post's social card: returns a 1200x630 PNG, published at //card.png. It draws the title in the headings' face, Xenon Bold ("MM Xe"), and "date · N min read" in the machine face, Krypton ("MM Kr"), in the dark fg and mute, onto a card base: the ruled header and an empty framed box, which scripts/card-base.sh draws from scripts/card-base.html, both sized by data/card.toml. images.Text has no width: it wraps only at the image's width less 20 and x, and keeps \n, so card-wrap.html wraps the title. A title too long for 3 lines at 48px fails the build */}} {{- /* a card.png in the post's bundle would be published at the card's address, and replace it or be replaced */}} {{- with .Resources.Get "card.png" }} {{- errorf "%s: the post's bundle has a card.png, which would take the social card's address; rename it" $.File.Path }} {{- end }} {{- $text := replaceRE `\s+` " " (strings.TrimSpace .Title) }} {{- /* images.Text draws a character its font lacks as an empty box, so a title with one fails the build. The card's TTFs keep the ranges in data/card_chars.toml */}} {{- $class := "" }} {{- range hugo.Data.card_chars.ranges }} {{- $class = print $class (delimit (apply (split . "-") "printf" "\\x{%s}" ".") "-") }} {{- end }} {{- with findRE (printf "[^%s]" $class) $text }} {{- errorf "%s: the social card's font can't draw %q in the title; change them, or add them to data/card_chars.toml and rerun scripts/fonts.py" $.File.Path (delimit (uniq .) "") }} {{- end }} {{- $words := split $text " " }} {{- $box := hugo.Data.card }} {{- $x := add $box.box_left $box.inset }} {{- $width := sub $box.box_width (mul 2 $box.inset) }} {{- $wrap := partial "card-wrap.html" (dict "words" $words "width" $width) }} {{- $size := $wrap.size }} {{- $cols := $wrap.cols }} {{- $lines := $wrap.lines }} {{- /* the smallest size's lines hold $cols characters, so a word longer than that never fits */}} {{- if not $size }} {{- $long := "" }} {{- range $words }}{{ if gt (strings.RuneCount .) $cols }}{{ $long = . }}{{ break }}{{ end }}{{ end }} {{- with $long }} {{- errorf "%s: the title's word %q is %d characters, too long for the social card's lines of %d; shorten it" $.File.Path . (strings.RuneCount .) $cols }} {{- else }} {{- errorf "%s: the title is too long for the social card, which fits 3 lines of %d characters; shorten it" $.File.Path $cols }} {{- end }} {{- return "" }} {{- end }} {{- /* The box fits the title: card-base-N.png's box holds N lines at 62px (data/card.toml). A title at 54 or 48px takes the smallest box that holds its lines, and the room left over goes half above the title and half below the meta line. The title's lines are ⌈1.13 × size⌉ apart (71px at 62, a line height of 1.14), and each baseline sits that far below the top of its line. images.Text puts the first baseline one ascent below y, rounded up, and each next one an ascent plus the line spacing further down. Monaspace's ascent is 0.945em, so the spacing makes up the difference, and the text is drawn that much lower. The meta line is placed the same way at 23px */}} {{- $pitch := int (math.Ceil (mul 1.13 $size)) }} {{- $gap := sub $pitch (int (math.Ceil (mul 0.945 $size))) }} {{- $metagap := sub (int (math.Ceil (mul 1.13 23))) (int (math.Ceil (mul 0.945 23))) }} {{- $height := mul (len $lines) $pitch }} {{- $n := div (add $height (sub $box.pitch 1)) $box.pitch }} {{- $top := div (sub $box.centre2 (add $box.base (mul $box.pitch $n))) 2 }} {{- $y := add $top $box.title_top (div (sub (mul $box.pitch $n) $height) 2) }} {{- /* images.Text keys a font by its path in the image cache, so a changed TTF would leave the cached cards as they were. Fingerprinting puts the font's hash in its path. Neither font is published */}} {{- $head := resources.Get "fonts/MMXe-Bold.ttf" | fingerprint }} {{- $machine := resources.Get "fonts/MMKr-Regular.ttf" | fingerprint }} {{- $palette := partialCached "palette.html" "palette" }} {{- $title := images.Text (delimit $lines "\n") (dict "font" $head "size" $size "linespacing" $gap "color" $palette.fg.dark "x" $x "y" (add $y $gap)) }} {{- $meta := images.Text (printf "%s · %d min read" (.Date.Format "2006-01-02") .ReadingTime) (dict "font" $machine "size" 23 "color" $palette.muted.dark "x" $x "y" (add $y $height $box.meta_gap $metagap)) }} {{- $card := resources.Get (printf "images/card-base-%d.png" $n) | images.Filter $title $meta }} {{- /* .RelPermalink is percent-encoded, and a target path is written as given, so the card takes the decoded path: a post at /café/ gets café/card.png, not caf%C3%A9/card.png */}} {{- return resources.Copy (path.Join (strings.TrimPrefix "/" (urls.Parse .RelPermalink).Path) "card.png") $card }}
