{{- /* Tag pages send readers to the index, where search replaced them: /tags/nixos/ goes to /#nixos and /tags/ to /. Only the HTML redirects, so a term's feed stays, and the stub links to it for feed readers, which don't follow the refresh. A term's title is its slug, #nixos, as the menu shows it, not Hugo's title-cased Nixos */}} {{- $to := "/" }} {{- $title := .Title }} {{- $label := "every post" }} {{- if eq .Kind "term" }} {{- $slug := partial "tag-slug.html" . }} {{- $to = print "/#" $slug }} {{- $title = print "#" $slug }} {{- $label = printf "posts tagged %s" $slug }} {{- end }} {{- partial "redirect.html" (dict "title" (printf "%s · %s" $title site.Title) "heading" $title "to" $to "canonical" site.Home.Permalink "label" $label "feed" (.OutputFormats.Get "atom")) }}
